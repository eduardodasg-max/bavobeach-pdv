const express = require('express'); const app = express(); app.use(express.json()); // Banco de dados em memória (Substitua por PostgreSQL/MongoDB em produção) let estoque = { "1": { id: "1", nome: "Cerveja Duplo Malte", tipo: "bebida", preco: 12.00, qtd: 50 }, "2": { id: "2", nome: "Batata Frita c/ Queijo", tipo: "petisco", preco: 35.00, qtd: 20 }, "3": { id: "3", nome: "Filé à Parmegiana", tipo: "prato", preco: 65.00, qtd: 15 } }; let mesas = {}; const TOTAL_MESAS = 15; // Inicializa o mapa de mesas livres for (let i = 1; i <= TOTAL_MESAS; i++) { mesas[i] = { numero: i, aberta: false, itens: [], taxaServicoPadrao: 10, desconto: 0 }; } // === ROTAS DE PRODUTOS E ESTOQUE === app.get('/produtos', (req, res) => { res.json(Object.values(estoque)); }); // === ROTAS DO MAPA DE MESAS === app.get('/mesas', (req, res) => { res.json(Object.values(mesas)); }); // Abrir ou lançar item na mesa (Mantém aberta e disponível) app.post('/mesas/:numero/lancar', (req, res) => { const { numero } = req.params; const { produtoId, quantidade } = req.body; const mesa = mesas[numero]; if (!mesa) return res.status(404).json({ error: "Mesa não encontrada." }); const produto = estoque[produtoId]; if (!produto) return res.status(404).json({ error: "Produto não encontrado." }); // Validação rigorosa de estoque para evitar falhas if (produto.qtd < quantidade) { return res.status(400).json({ error: `Estoque insuficiente. Disponível: ${produto.qtd}` }); } // Deduz do estoque imediatamente produto.qtd -= quantidade; // Se a mesa estava fechada, ela abre agora if (!mesa.aberta) { mesa.aberta = true; mesa.itens = []; mesa.desconto = 0; mesa.taxaServicoPadrao = 10; } // Adiciona o item à mesa const itemExistente = mesa.itens.find(i => i.produtoId === produtoId); if (itemExistente) { itemExistente.quantidade += quantidade; } else { mesa.itens.push({ produtoId, nome: produto.nome, precoUnitario: produto.preco, quantidade }); } res.json({ message: "Item lançado com sucesso!", mesa }); }); // Cancelamento de Item (Estorno de estoque) app.post('/mesas/:numero/cancelar-item', (req, res) => { const { numero } = req.params; const { produtoId, quantidade } = req.body; const mesa = mesas[numero]; if (!mesa || !mesa.aberta) return res.status(400).json({ error: "Mesa não está ativa." }); const item = mesa.itens.find(i => i.produtoId === produtoId); if (!item || item.quantidade < quantidade) { return res.status(400).json({ error: "Quantidade de cancelamento inválida." }); } // Estorna ao estoque item.quantidade -= quantity; estoque[produtoId].qtd += quantidade; // Limpa o item se a quantidade zerar if (item.quantidade === 0) { mesa.itens = mesa.itens.filter(i => i.produtoId !== produtoId); } res.json({ message: "Item cancelado e estoque estornado.", mesa }); }); // Atualizar taxas e descontos editáveis da mesa app.put('/mesas/:numero/configurar-valores', (req, res) => { const { numero } = req.params; const { taxaServico, desconto } = req.body; const mesa = mesas[numero]; if (!mesa || !mesa.aberta) return res.status(400).json({ error: "Mesa não ativa." }); if (taxaServico !== undefined) mesa.taxaServicoPadrao = taxaServico; if (desconto !== undefined) mesa.desconto = desconto; res.json({ message: "Valores atualizados com sucesso.", mesa }); }); // Fechamento e Encerramento da Mesa app.post('/mesas/:numero/fechar', (req, res) => { const { numero } = req.params; const mesa = mesas[numero]; if (!mesa || !mesa.aberta) return res.status(400).json({ error: "Mesa já está fechada ou não existe." }); // Cálculo dos totais const subtotal = mesa.itens.reduce((acc, item) => acc + (item.precoUnitario * item.quantidade), 0); const valorTaxa = subtotal * (mesa.taxaServicoPadrao / 100); const totalFinal = Math.max(0, subtotal + valorTaxa - mesa.desconto); const resumoFechamento = { numeroMesa: mesa.numero, itens: mesa.itens, subtotal, taxaServico: `${mesa.taxaServicoPadrao}% (${valorTaxa.toFixed(2)})`, desconto: mesa.desconto, total: totalFinal }; // Reseta a mesa para o estado Livre mesa.aberta = false; mesa.itens = []; mesa.desconto = 0; res.json({ message: "Mesa encerrada com sucesso!", cupomFiscal: resumoFechamento }); }); app.listen(3000, () => console.log("Servidor PDV rodando na porta 3000"));
